# 1指针加减整数
```c
#define n 5
float values[n];
float *vp;

//指针加减整数
for (vp = &values[0], vp < values[n]; )//无需步骤直接继续循环
{
	*vp++ = 0;//vp = 0; vp++;(后置++)
	
	}//程序相当于给values【0】->values[n]都赋值为0
 *vp++为给地址++
 （*vp）++为给对象++
```

## 用法
```c
strlen的自我实现
1.判断\0
int my_strlen(char* str){
	int count = 0;
	while(*str != '\0')
		count++;
		str++;
	}
	return count;
}

2.指针-指针
int my_strlen(char* str){
	char* start = str;
	while(*str != '\0'){
		str++;
		}
	return （str - start）;
	}
```
