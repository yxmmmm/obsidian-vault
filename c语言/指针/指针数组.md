int main(){
	int a = 10;
	int b = 20;
	int c = 30;
	
	int arr[10];
	int *pa = &a;
	int *pb = &b;
	int *pc = &c;
	
	//parr为存放指针的数组--指针数组
	int* parr[10] = {&a, &b, &c};
	
	for(int i = 0; i < 3; i++)
	{
		printf("%d ", *(parr[i]));//输出10 20 30
	}
	
	//使用指针数组模拟二维数组
	int arr1[4] = {1,2,3,4};
	int arr2[4] = {2,3,4,5};
	int arr3[4] = {3,4,5,6};
	
	int* parr[3] = {arr1, arr2, arr3};
	for(int i = 0; i < 3; i++){
		for(int j = 0; j < 4; j++){
			printf("%d", parr[i][j]);
			//C 语言中，parr[i][j] 会被编译器解释为：*(*(parr + i) + j)
			//- 第一步：parr + i 得到第 i 个指针的地址，*(parr + i) 解引用得到该指针（即arr1、arr2 或 arr3 的首地址）。
		//- 第二步：*(parr + i) + j 得到该指针指向的数组中第 j 个元素的地址，再通过外层的 * 解引用得到该元素的值。
		}
	}
	return 0；
}
