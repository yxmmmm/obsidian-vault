const calendarData = {
    year: 2025,
    colors: {
        blue: ["#ebedf0", "#c6e48b", "#7bc96f", "#239a3b", "#196127"], // 复刻 GitHub 绿白阶梯
    },
    showCurrentDayBorder: true,
    boxSize: 11,      // 关键：设定格子大小，11-12 是最像正方形的尺寸
    entries: []
}

// 扫描全库所有笔记
for (let page of dv.pages()) {
    // 自动抓取文件的创建时间作为日期
    // 这样即便你以前没写日期标签，也能统计到
    const dateStr = page.file.ctime.toFormat("yyyy-MM-dd");
    
    calendarData.entries.push({
        date: dateStr,
        intensity: 1, 
        content: "" 
    })
}

	renderHeatmapCalendar(this.container, calendarData)
