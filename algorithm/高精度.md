## 代码实现

```cpp
#include <iostream>

using namespace std;

const int N = 1e6 + 10;

// C=A+B
vector<int> add(vector<int> &A, vector<int> &B)
{
    vector<int> C;

    int t = 0;//进位
    for (int i = 0; i < A.size() || i < B.size(); i++)
    {
        if (i < A.size())
            t += A[i];
        if (i < B.size())
            t += B[i];
        C.push_back(t % 10);
        t /= 10;
    }
    if (t)
        C.push_back(1);
    return C;
}

int main()
{
    string a, b;
    vector<int> A, B;

    cin >> a >> b; // a="123456"
    for (int i = a.size() - 1; i >= 0; i--)
        A.push_back(a[i] - '0'); // A=[6,5,4,3,2,1]
    for (int i = b.size() - 1; i >= 0; i--)
        B.push_back(b[i] - '0');

    auto C = add(A, B);

    for (int i = C.size() - 1; i >= 0; i--)
        cout << C[i];
}

```

# 高精度减法

## 思路及方法

![](https://tuchuang-1387570672.cos.ap-nanjing.myqcloud.com/obsidianvault%E7%AE%97%E6%B3%95-7.jpg)

两个数相减可转化为两个数的绝对值相加减的形式

## 代码实现

```cpp
#include <iostream>

using namespace std;

const int N = 1e6 + 10;

// 判断A是否>=B
bool cmp(vector<int> &A, vector<int> &B)
{
    if (A.size() != B.size())
        return A.size() > B.size();
    for (int i = A.size()-1; i >= 0; i++)
    {
        if (A[i] != B[i])
            return A[i] > B[i];
    }
    return true;
}

// C=A-B
vector<int> sub(vector<int> &A, vector<int> &B)
{
    vector<int> C;
    for (int i = 0, t = 0; i < A.size(); i++)
    {
        t = A[i] - t;
        if (i < B.size())
            t -= B[i];
        C.push_back((t + 10) % 10);
        if (t < 0)
            t = 1;
        else
            t = 0;
    }

    while (C.size() > 1 && C.back() == 0)
        C.pop_back();//去掉前导0

    return C;
}

int main()
{
    string a, b;
    vector<int> A, B;

    cin >> a >> b; // a="123456"
    for (int i = a.size() - 1; i >= 0; i--)
        A.push_back(a[i] - '0'); // A=[6,5,4,3,2,1]
    for (int i = b.size() - 1; i >= 0; i--)
        B.push_back(b[i] - '0');

    if (cmp(A, B))
    {
        auto C = sub(A, B);

        for (int i = C.size() - 1; i >= 0; i--)
            cout << C[i];
    }
}

```



核心步骤拆解：

1. **逐位相减**：`t = A[i] - t`。这里的 `t` 是上一位留下的借位。

2. **减去 B**：如果 $B$ 在这一位还有数字，就减去它：`t -= B[i]`。

3. **存入结果**：`(t + 10) % 10`。

   - 如果 `t >= 0`：比如 $7 - 3 = 4$，$(4+10)\%10 = 4$，没毛病。
   - 如果 `t < 0`：比如 $3 - 7 = -4$，$(-4+10)\%10 = 6$，这正是借位后的结果。

4. **更新借位**：

   - 如果 `t < 0`，说明不够减，向高位借 1，所以下一次循环 `t = 1`。
   - 否则 `t = 0`。

   

(t+10)%10可以实现
\[
f(t) = \begin{cases}
t, & t \ge 0 \\
t + 10, & t < 0
\end{cases}
\]
