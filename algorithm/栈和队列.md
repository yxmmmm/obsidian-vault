# 模拟栈和队列

栈-先进后出

队列-先进先出

```cpp
#include <iostream>
using namespace std;
const int N = 1e6 + 10;

/*-------------stack----------------*/
int stk[N], tt = 0;

// insert
stk[++tt] = x;

// remove
tt--;

// ifempty
if (tt > 0)
    not empty else empty

        // top
        stk[tt];

/*-------------queue----------------*/

// 队尾插入元素队头弹出元素hh队头，tt队尾
int q[N], hh = 0, tt = -1;

// insert
q[++tt] = x;

// 弹出
h++; // hh向后移动一位相当于队头弹出。hh------tt

// ifempty
if (hh <= tt)
    not empty 
else empty

// head
q[hh];

// tail
q[tt];
```

# 单调栈

![IMG_38822047D2D4-1](https://tuchuang-1387570672.cos.ap-nanjing.myqcloud.com/obsidianvaultIMG_38822047D2D4-1.jpeg)

```cpp
#include <iostream>

using namespace std;

const int N = 100010;

int stk[N], tt;

int main()
{
    int n;
    cin >> n;
    while (n -- )
    {
        int x;
        scanf("%d", &x);
        while (tt && stk[tt] >= x) tt -- ;//tt=0或实现单调结束
        if (!tt) printf("-1 ");//没有满足情况
        else printf("%d ", stk[tt]);//返回最近数
        stk[ ++ tt] = x;//进栈下一元素
    }

    return 0;
}
```

# 单调队列

![算法-21](https://tuchuang-1387570672.cos.ap-nanjing.myqcloud.com/obsidianvault%E7%AE%97%E6%B3%95-21.jpg)

![IMG_70AEA3EACC10-1](https://tuchuang-1387570672.cos.ap-nanjing.myqcloud.com/obsidianvaultIMG_70AEA3EACC10-1.jpeg)

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e6 + 10;
int n, k;
int a[N], q[N];

int main()
{
    scanf("%d%d", &n, &k);
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    int hh = 0, tt = -1;
    for (int i = 0; i < n; i++)
    {
        if (hh <= tt && i - k + 1 > q[hh])
            hh++; // hh<=tt为了满足q中存在元素，i-k+1>q[hh]判断窗口左边界是否大于最小数从而决定是否踢掉老数

        while (hh <= tt && a[q[tt]] >= a[i])
            tt--;    // 当新数<老数，踢掉老数直到单调
        q[++tt] = i; // 补新数

        if (i >= k - 1)
            printf("%d ", a[q[hh]]);
    }
    puts("");
    
    //反转
    hh = 0, tt = -1;
    for (int i = 0; i < n; i++)
    {
        if (hh <= tt && i - k + 1 > q[hh])
            hh++;

        while (hh <= tt && a[q[tt]] <= a[i])
            tt--;
        q[++tt] = i;

        if (i >= k - 1)
            printf("%d ", a[q[hh]]);
    }
}
```
